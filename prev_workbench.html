sas_program_tokenization.sas 
/* ===================================================================== */
/* PROGRAM 1: Free Text Tokenization, Stopword Removal, & N-Gram Generation */
/* Environment: Standard SAS Enterprise Guide (No Add-ons Required)     */
/* ===================================================================== */

data work.complaint_tokens;
    set work.raw_complaints;
    
    /* Clean text: lowercase, remove special characters and punctuation */
    clean_desc = lowcase(compress(customer_description, ., 'ai'));
    clean_findings = lowcase(compress(investigation_findings, ., 'ai'));
    
    /* Combine with dual-field weighting (70% findings, 30% description) */
    /* In actual SAS EG, we tokenize both independently or concatenate weighted */
    combined_text = catx(' ', clean_findings, clean_findings, clean_desc);
    
    word_count = countw(combined_text);
    
    /* 1. Unigrams Extraction & Stopword Filter */
    do i = 1 to word_count;
        term = scan(combined_text, i, ' ');
        gram_type = '1-gram';
        
        /* Inline Stopword Filtering */
        if not missing(term) and length(term) > 2 
           and term not in ('the','and','was','for','that','with','this','not','you','have','are','was') then do;
            output;
        end;
    end;
    
    /* 2. Bigrams Extraction */
    do i = 1 to (word_count - 1);
        term = catx(' ', scan(combined_text, i, ' '), scan(combined_text, i+1, ' '));
        gram_type = '2-gram';
        if not missing(term) then output;
    end;
    
    keep complaint_id term gram_type customer_id case_date;
run;
